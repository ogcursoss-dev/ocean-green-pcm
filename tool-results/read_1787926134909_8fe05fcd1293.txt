   246→function ExamQuestionsTab({
   247→  examId,
   248→  examQuestions,
   249→  onChanged,
   250→}: {
   251→  examId: string;
   252→  examQuestions: ExamData["questions"];
   253→  onChanged: () => void;
   254→}) {
   255→  const [subjects, setSubjects] = useState<Array<{ id: string; name: string }>>([]);
   256→  const [subjectFilter, setSubjectFilter] = useState("ALL");
   257→  const [diffFilter, setDiffFilter] = useState("ALL");
   258→  const [search, setSearch] = useState("");
   259→  const [results, setResults] = useState<QuestionSearch[]>([]);
   260→  const [loadingSearch, setLoadingSearch] = useState(false);
   261→  const [busy, setBusy] = useState(false);
   262→
   263→  useEffect(() => {
   264→    apiFetch<{ subjects: Array<{ id: string; name: string }> }>("/api/admin/subjects").then(
   265→      (r) => {
   266→        if (r.ok && r.data) setSubjects(r.data.subjects || []);
   267→      }
   268→    );
   269→  }, []);
   270→
   271→  useEffect(() => {
   272→    setLoadingSearch(true);
   273→    const t = setTimeout(async () => {
   274→      const params = new URLSearchParams();
   275→      if (search) params.set("search", search);
   276→      if (subjectFilter !== "ALL") params.set("subjectId", subjectFilter);
   277→      if (diffFilter !== "ALL") params.set("difficulty", diffFilter);
   278→      params.set("pageSize", "20");
   279→      const res = await apiFetch<{ questions: QuestionSearch[] }>(
   280→        `/api/admin/questions?${params.toString()}`
   281→      );
   282→      if (res.ok && res.data) setResults(res.data.questions || []);
   283→      setLoadingSearch(false);
   284→    }, 350);
   285→    return () => clearTimeout(t);
   286→  }, [search, subjectFilter, diffFilter]);
   287→
   288→  const examQuestionIds = new Set(examQuestions.map((eq) => eq.question.id));
   289→
   290→  async function addQuestion(qid: string) {
   291→    setBusy(true);
   292→    const res = await apiFetch(`/api/admin/exams/${examId}/questions`, {
   293→      method: "POST",
   294→      body: JSON.stringify({ questionId: qid }),
   295→    });
   296→    setBusy(false);
   297→    if (res.ok) {
   298→      toast.success("Questão adicionada.");
   299→      onChanged();
   300→    } else {
   301→      toast.error(res.error || "Erro ao adicionar.");
   302→    }
   303→  }
   304→
   305→  async function removeQuestion(qid: string) {
   306→    if (!confirm("Remover esta questão da prova?")) return;
   307→    setBusy(true);
   308→    const res = await apiFetch(`/api/admin/exams/${examId}/questions/${qid}`, {
   309→      method: "DELETE",
   310→    });
   311→    setBusy(false);
   312→    if (res.ok) {
   313→      toast.success("Questão removida.");
   314→      onChanged();
   315→    } else {
   316→      toast.error(res.error || "Erro ao remover.");
   317→    }
   318→  }
   319→
   320→  async function move(qid: string, direction: -1 | 1) {
   321→    const idx = examQuestions.findIndex((eq) => eq.question.id === qid);
   322→    if (idx < 0) return;
   323→    const newIdx = idx + direction;
   324→    if (newIdx < 0 || newIdx >= examQuestions.length) return;
   325→    setBusy(true);
   326→    await apiFetch(`/api/admin/exams/${examId}/questions/${qid}`, {
   327→      method: "PATCH",
   328→      body: JSON.stringify({ order: examQuestions[newIdx].order }),
   329→    });
   330→    setBusy(false);
   331→    onChanged();
   332→  }
   333→
   334→  return (
   335→    <div className="grid gap-4 lg:grid-cols-2">
   336→      {/* Lista de questões na prova */}
   337→      <Card>
   338→        <CardHeader>
   339→          <CardTitle className="text-base">
   340→            Questões da Prova ({examQuestions.length})
   341→          </CardTitle>
   342→        </CardHeader>
   343→        <CardContent className="space-y-2 max-h-[600px] overflow-y-auto">
   344→          {examQuestions.length === 0 ? (
   345→            <div className="py-8 text-center text-sm text-muted-foreground">
   346→              <FileText className="size-6 mx-auto mb-2 opacity-50" />
   347→              Nenhuma questão adicionada ainda.
   348→            </div>
   349→          ) : (
   350→            examQuestions.map((eq, idx) => (
   351→              <motion.div
   352→                key={eq.id}
   353→                initial={{ opacity: 0 }}
   354→                animate={{ opacity: 1 }}
   355→                className="flex items-start gap-2 p-3 rounded-lg border border-border bg-card hover:border-accent/40 transition-colors"
   356→              >
   357→                <div className="flex flex-col items-center">
   358→                  <GripVertical className="size-3.5 text-muted-foreground/50" />
   359→                  <span className="text-xs font-bold text-primary mt-1">
   360→                    {idx + 1}
   361→                  </span>
   362→                </div>
   363→                <div className="flex-1 min-w-0">
   364→                  <p className="text-sm line-clamp-2">
   365→                    {eq.question.statement}
   366→                  </p>
   367→                  <div className="flex flex-wrap gap-1 mt-1">
   368→                    {eq.question.subject && (
   369→                      <Badge variant="outline" className="text-xs">
   370→                        {eq.question.subject.name}
   371→                      </Badge>
   372→                    )}
   373→                    <Badge variant="outline" className="text-xs">
   374→                      {eq.question.difficulty === "EASY"
   375→                        ? "Fácil"
   376→                        : eq.question.difficulty === "MEDIUM"
   377→                        ? "Média"
   378→                        : "Difícil"}
   379→                    </Badge>
   380→                    <Badge variant="outline" className="text-xs font-mono">
   381→                      Gab: {eq.question.correctAnswer}
   382→                    </Badge>
   383→                  </div>
   384→                </div>
   385→                <div className="flex flex-col gap-0.5">
   386→                  <Button
   387→                    size="icon"
   388→                    variant="ghost"
   389→                    className="size-7"
   390→                    disabled={busy || idx === 0}
   391→                    onClick={() => move(eq.question.id, -1)}
   392→                    title="Subir"
   393→                  >
   394→                    <ArrowUp className="size-3.5" />
   395→                  </Button>
   396→                  <Button
   397→                    size="icon"
   398→                    variant="ghost"
   399→                    className="size-7"
   400→                    disabled={busy || idx === examQuestions.length - 1}
   401→                    onClick={() => move(eq.question.id, 1)}
   402→                    title="Descer"
   403→                  >
   404→                    <ArrowDown className="size-3.5" />
   405→                  </Button>
   406→                </div>
   407→                <Button
   408→                  size="icon"
   409→                  variant="ghost"
   410→                  className="size-7 text-destructive hover:text-destructive"
   411→                  disabled={busy}
   412→                  onClick={() => removeQuestion(eq.question.id)}
   413→                  title="Remover"
   414→                >
   415→                  <Trash2 className="size-3.5" />
   416→                </Button>
   417→              </motion.div>
   418→            ))
   419→          )}
   420→        </CardContent>
   421→      </Card>
   422→
   423→      {/* Buscar e adicionar */}
   424→      <Card>
   425→        <CardHeader>
   426→          <CardTitle className="text-base flex items-center gap-2">
   427→            <Plus className="size-4 text-accent" />
   428→            Adicionar Questões
   429→          </CardTitle>
   430→        </CardHeader>
   431→        <CardContent className="space-y-3">
   432→          <div className="relative">
   433→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   434→            <Input
   435→              placeholder="Buscar por enunciado..."
   436→              value={search}
   437→              onChange={(e) => setSearch(e.target.value)}
   438→              className="pl-9"
   439→            />
   440→          </div>
   441→          <div className="grid grid-cols-2 gap-2">
   442→            <Select value={subjectFilter} onValueChange={setSubjectFilter}>
   443→              <SelectTrigger>
   444→                <SelectValue placeholder="Disciplina" />
   445→              </SelectTrigger>
   446→              <SelectContent className="max-h-72">
   447→                <SelectItem value="ALL">Todas</SelectItem>
   448→                {subjects.map((s) => (
   449→                  <SelectItem key={s.id} value={s.id}>
   450→                    {s.name}
   451→                  </SelectItem>
   452→                ))}
   453→              </SelectContent>
   454→            </Select>
   455→            <Select value={diffFilter} onValueChange={setDiffFilter}>
   456→              <SelectTrigger>
   457→                <SelectValue placeholder="Dificuldade" />
   458→              </SelectTrigger>
   459→              <SelectContent>
   460→                <SelectItem value="ALL">Todas</SelectItem>
   461→                <SelectItem value="EASY">Fácil</SelectItem>
   462→                <SelectItem value="MEDIUM">Média</SelectItem>
   463→                <SelectItem value="HARD">Difícil</SelectItem>
   464→              </SelectContent>
   465→            </Select>
   466→          </div>
   467→          <div className="border rounded-lg max-h-[420px] overflow-y-auto divide-y divide-border">
   468→            {loadingSearch ? (
   469→              <div className="p-4 text-center text-sm text-muted-foreground">
   470→                <Loader2 className="size-4 animate-spin mx-auto" />
   471→              </div>
   472→            ) : results.length === 0 ? (
   473→              <div className="p-4 text-center text-sm text-muted-foreground">
   474→                Nenhuma questão encontrada.
   475→              </div>
   476→            ) : (
   477→              results.map((q) => {
   478→                const inExam = examQuestionIds.has(q.id);
   479→                return (
   480→                  <div
   481→                    key={q.id}
   482→                    className="flex items-start gap-2 p-3 hover:bg-muted/40"
   483→                  >
   484→                    <div className="flex-1 min-w-0">
   485→                      <p className="text-sm line-clamp-2">{q.statement}</p>
   486→                      <div className="flex gap-1 mt-1 flex-wrap">
   487→                        {q.subject && (
   488→                          <Badge variant="outline" className="text-xs">
   489→                            {q.subject.name}
   490→                          </Badge>
   491→                        )}
   492→                        <Badge variant="outline" className="text-xs">
   493→                          {q.difficulty === "EASY"
   494→                            ? "Fácil"
   495→                            : q.difficulty === "MEDIUM"
   496→                            ? "Média"
   497→                            : "Difícil"}
   498→                        </Badge>
   499→                      </div>
   500→                    </div>
   501→                    <Button
   502→                      size="sm"
   503→                      variant={inExam ? "secondary" : "default"}
   504→                      disabled={inExam || busy}
   505→                      onClick={() => addQuestion(q.id)}
   506→                    >
   507→                      {inExam ? (
   508→                        <>
   509→                          <X className="size-3 mr-1" />
   510→                          Adicionada
   511→                        </>
   512→                      ) : (
   513→                        <>
   514→                          <Plus className="size-3 mr-1" />
   515→                          Adicionar
   516→                        </>
   517→                      )}
   518→                    </Button>
   519→                  </div>
   520→                );
   521→              })
   522→            )}
   523→          </div>
   524→        </CardContent>
   525→      </Card>
   526→    </div>
   527→  );
   528→}

   530→function ExamAssignmentsTab({
   531→  examId,
   532→  assignments,
   533→  classId,
   534→  onChanged,
   535→}: {
   536→  examId: string;
   537→  assignments: ExamData["assignments"];
   538→  classId?: string;
   539→  onChanged: () => void;
   540→}) {
   541→  const [addOpen, setAddOpen] = useState(false);
   542→  const [members, setMembers] = useState<ClassMember[]>([]);
   543→  const [membersLoading, setMembersLoading] = useState(false);
   544→  const [target, setTarget] = useState<"CLASS" | "INDIVIDUAL">("CLASS");
   545→  const [selectedUserId, setSelectedUserId] = useState("");
   546→  const [useCustom, setUseCustom] = useState(false);
   547→  const [customStart, setCustomStart] = useState("");
   548→  const [customEnd, setCustomEnd] = useState("");
   549→  const [customDuration, setCustomDuration] = useState("");
   550→  const [notes, setNotes] = useState("");
   551→  const [saving, setSaving] = useState(false);
   552→
   553→  async function loadMembers() {
   554→    if (!classId) return;
   555→    setMembersLoading(true);
   556→    const res = await apiFetch<{ members: ClassMember[] }>(
   557→      `/api/admin/classes/${classId}/members`
   558→    );
   559→    if (res.ok && res.data) setMembers(res.data.members || []);
   560→    setMembersLoading(false);
   561→  }
   562→
   563→  useEffect(() => {
   564→    if (addOpen) loadMembers();
   565→  }, [addOpen]);
   566→
   567→  async function handleCreate() {
   568→    if (target === "INDIVIDUAL" && !selectedUserId) {
   569→      toast.error("Selecione um aluno.");
   570→      return;
   571→    }
   572→    setSaving(true);
   573→    const body: any = {
   574→      userId: target === "INDIVIDUAL" ? selectedUserId : null,
   575→      classId,
   576→      notes: notes || undefined,
   577→    };
   578→    if (useCustom) {
   579→      if (customStart) body.customStart = new Date(customStart).toISOString();
   580→      if (customEnd) body.customEnd = new Date(customEnd).toISOString();
   581→      if (customDuration) body.customDuration = Number(customDuration);
   582→    }
   583→    const res = await apiFetch(`/api/admin/exams/${examId}/assignments`, {
   584→      method: "POST",
   585→      body: JSON.stringify(body),
   586→    });
   587→    setSaving(false);
   588→    if (!res.ok) {
   589→      toast.error(res.error || "Erro ao criar atribuição.");
   590→      return;
   591→    }
   592→    toast.success("Atribuição criada.");
   593→    setAddOpen(false);
   594→    setUseCustom(false);
   595→    setCustomStart("");
   596→    setCustomEnd("");
   597→    setCustomDuration("");
   598→    setNotes("");
   599→    setSelectedUserId("");
   600→    onChanged();
   601→  }
   602→
   603→  async function handleDelete(assignmentId: string) {
   604→    if (!confirm("Remover esta atribuição?")) return;
   605→    const res = await apiFetch(
   606→      `/api/admin/exams/${examId}/assignments/${assignmentId}`,
   607→      { method: "DELETE" }
   608→    );
   609→    if (res.ok) {
   610→      toast.success("Atribuição removida.");
   611→      onChanged();
   612→    } else {
   613→      toast.error(res.error || "Erro ao remover.");
   614→    }
   615→  }
   616→
   617→  return (
   618→    <div className="space-y-3">
   619→      <div className="flex justify-end">
   620→        <Button onClick={() => setAddOpen(true)}>
   621→          <UserPlus className="size-4 mr-1" />
   622→          Nova Atribuição
   623→        </Button>
   624→      </div>
   625→
   626→      <Card>
   627→        <CardHeader>
   628→          <CardTitle className="text-base">
   629→            Atribuições da Prova ({assignments.length})

   631→          <CardDescription>

   653→                    className={cn(
   654→                      "size-10 rounded-lg flex items-center justify-center",

   685→                          <Calendar className="size-3" />

   687→                          {a.customStart
   688→                            ? formatDateTime(a.customStart)
   689→                            : "—"}{" "}
   690→                          →{" "}
   691→                          {a.customEnd ? formatDateTime(a.customEnd) : "—"}
   692→                        </span>
   693→                      ) : (
   694→                        <span className="flex items-center gap-1">
   695→                          <Clock className="size-3" />
   696→                          Usa janela padrão da prova
   697→                        </span>
   698→                      )}
   699→                      {a.customDuration && (
   700→                        <span className="flex items-center gap-1">
   701→                          <Clock className="size-3" />
   702→                          Duração: {a.customDuration} min
   703→                        </span>
   704→                      )}
   705→                    </div>
   706→                    {a.notes && (
   707→                      <p className="text-xs text-muted-foreground mt-1 italic">
   708→                        "{a.notes}"
   709→                      </p>
   710→                    )}
   711→                  </div>
   712→                  <Button
   713→                    size="icon"
   714→                    variant="ghost"
   715→                    className="size-8 text-destructive hover:text-destructive"
   716→                    onClick={() => handleDelete(a.id)}
   717→                    title="Remover"
   718→                  >
   719→                    <Trash2 className="size-4" />
   720→                  </Button>
   721→                </div>
   722→              ))}
   723→            </div>
   724→          )}
   725→        </CardContent>
   726→      </Card>
   727→
   728→      {/* Dialog nova atribuição */}
   729→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   730→        <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   731→          <DialogHeader>
   732→            <DialogTitle>Nova Atribuição</DialogTitle>
   733→            <DialogDescription>
   734→              Atribua a prova à turma inteira ou a um aluno específico (prova
   735→              antecipada individual).
   736→            </DialogDescription>
   737→          </DialogHeader>
   738→          <div className="space-y-4">
   739→            <div className="grid gap-2">

   764→                    <AlertCircle className="size-4 inline mr-1" />

   766→                  </div>

   768→                  <Select value={selectedUserId} onValueChange={setSelectedUserId}>
   769→                    <SelectTrigger>
   770→                      <SelectValue placeholder="Selecione o aluno" />

   772→                    <SelectContent>
   773→                      {members.map((m) => (

   783→            <div className="flex items-center gap-2">
   784→              <Switch

   797→                  <Input
   798→                    id="cs"
   799→                    type="datetime-local"
   800→                    value={customStart}
   801→                    onChange={(e) => setCustomStart(e.target.value)}
   802→                  />
   803→                </div>
   804→                <div className="grid gap-2">
   805→                  <Label htmlFor="ce">Fim custom.</Label>
   806→                  <Input
   807→                    id="ce"
   808→                    type="datetime-local"
   809→                    value={customEnd}
   810→                    onChange={(e) => setCustomEnd(e.target.value)}
   811→                  />
   812→                </div>
   813→                <div className="grid gap-2">
   814→                  <Label htmlFor="cd">Duração (min)</Label>
   815→                  <Input
   816→                    id="cd"
   817→                    type="number"
   818→                    min="1"
   819→                    value={customDuration}
   820→                    onChange={(e) => setCustomDuration(e.target.value)}
   821→                  />
   822→                </div>
   823→                <p className="text-xs text-muted-foreground md:col-span-3">
   824→                  Deixe em branco para usar o padrão da prova.
   825→                </p>
   826→              </div>
   827→            )}
   828→            <div className="grid gap-2">
   829→              <Label htmlFor="notes">Observações</Label>
   830→              <Textarea
   831→                id="notes"
   832→                value={notes}

   834→                rows={2}

   837→            </div>
   838→          </div>
   839→          <DialogFooter>
   840→            <Button variant="outline" onClick={() => setAddOpen(false)}>

   845→              Criar atribuição
   846→            </Button>
   847→          </DialogFooter>
   848→        </DialogContent>
